pkg install -y inotify-tools  


cat > ~/git_watcher.sh <<'EOF'
#!/data/data/com.termux/files/usr/bin/bash
while inotifywait -e modify ~/se1; do
  cd ~/se1 && \
  git add . && \
  git commit -m "Auto-commit $(date)" && \
  git push
done
EOF

chmod +x ~/git_watcher.sh
nohup bash ~/git_watcher.sh > ~/watcher.log 2>&1 &
tail -f ~/watcher.log

echo "Test" >> ~/se1/README.md
